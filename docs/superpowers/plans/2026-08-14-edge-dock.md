```ts
import { NOTE_COLLAPSED_HEIGHT } from './note-window';

export const NOTE_DOCK_WIDTH = 8;
export const NOTE_DOCK_HEIGHT = 56;
export const NOTE_DOCK_EDGE_THRESHOLD_PX = 24;
export const NOTE_DOCK_UNFOLD_THRESHOLD_PX = 48;
export const NOTE_DOCK_STACK_GAP_PX = 8;
export const NOTE_DOCK_MIN_GRAB_PX = 24;

export type DockSide = 'left' | 'right';

export type Rect = {
  x: number;
  y: number;
  width: number;
  height: number;
};

export type DisplayWorkArea = {
  x: number;
  y: number;
  width: number;
  height: number;
};

export function resolveCollapsedDockSide(
  bar: Rect,
  workArea: DisplayWorkArea
): DockSide | undefined {
  const leftDistance = bar.x - workArea.x;
  const rightDistance = workArea.x + workArea.width - (bar.x + bar.width);
  if (leftDistance <= NOTE_DOCK_EDGE_THRESHOLD_PX && leftDistance <= rightDistance) {
    return 'left';
  }
  if (rightDistance <= NOTE_DOCK_EDGE_THRESHOLD_PX) {
    return 'right';
  }
  return undefined;
}

export function resolveDockedRelease(input: {
  side: DockSide;
  origin: Rect;
  current: Rect;
}): 'expand' | 'snap-back' {
  const delta =
    input.side === 'left' ? input.current.x - input.origin.x : input.origin.x - input.current.x;
  return delta >= NOTE_DOCK_UNFOLD_THRESHOLD_PX ? 'expand' : 'snap-back';
}

export function buildDockedBounds(input: {
  side: DockSide;
  y: number;
  workArea: DisplayWorkArea;
}): Rect {
  const y = clamp(
    input.y,
    input.workArea.y,
    input.workArea.y + input.workArea.height - NOTE_DOCK_HEIGHT
  );
  return {
    x:
      input.side === 'left'
        ? input.workArea.x
        : input.workArea.x + input.workArea.width - NOTE_DOCK_WIDTH,
    y,
    width: NOTE_DOCK_WIDTH,
    height: NOTE_DOCK_HEIGHT
  };
}

export function buildExpandBoundsFromDock(input: {
  side: DockSide;
  sliver: Rect;
  expandedSize: { width: number; height: number };
  workArea: DisplayWorkArea;
}): Rect {
  const width = Math.min(input.expandedSize.width, input.workArea.width);
  const height = Math.min(input.expandedSize.height, input.workArea.height);
  const x =
    input.side === 'left' ? input.sliver.x : input.sliver.x + input.sliver.width - width;
  const y = input.sliver.y;
  return {
    x: clamp(x, input.workArea.x, input.workArea.x + input.workArea.width - width),
    y: clamp(y, input.workArea.y, input.workArea.y + input.workArea.height - height),
    width,
    height
  };
}

export function offsetDockYToAvoidOverlap(input: {
  y: number;
  side: DockSide;
  workArea: DisplayWorkArea;
  occupied: Array<{ side: DockSide; y: number }>;
}): number {
  const taken = input.occupied
    .filter((item) => item.side === input.side)
    .map((item) => item.y)
    .sort((a, b) => a - b);
  const minY = input.workArea.y;
  const maxY = input.workArea.y + input.workArea.height - NOTE_DOCK_HEIGHT;
  let y = clamp(input.y, minY, maxY);
  const overlaps = (candidate: number): boolean =>
    taken.some((existing) => Math.abs(existing - candidate) < NOTE_DOCK_HEIGHT);
  while (overlaps(y) && y + NOTE_DOCK_HEIGHT + NOTE_DOCK_STACK_GAP_PX <= maxY) {
    y += NOTE_DOCK_HEIGHT + NOTE_DOCK_STACK_GAP_PX;
  }
  while (overlaps(y) && y - NOTE_DOCK_HEIGHT - NOTE_DOCK_STACK_GAP_PX >= minY) {
    y -= NOTE_DOCK_HEIGHT + NOTE_DOCK_STACK_GAP_PX;
  }
  if (overlaps(y)) {
    const last = taken[taken.length - 1] ?? y;
    y = clamp(last + NOTE_DOCK_MIN_GRAB_PX, minY, maxY);
  }
  return y;
}

function clamp(value: number, min: number, max: number): number {
  return Math.min(Math.max(value, min), max);
}
```

function normalizeNoteDock(value: unknown): NoteDock | undefined {
  if (!value || typeof value !== 'object') {
    return undefined;
  }
  const candidate = value as { side?: unknown; y?: unknown };
  if (candidate.side !== 'left' && candidate.side !== 'right') {
    return undefined;

  if (typeof candidate.y !== 'number' || !Number.isFinite(candidate.y)) {
    return undefined;

  return { side: candidate.side, y: candidate.y };
}

### Task 3: 窗口控制器第三态

**Files:**
- Modify: `main/note-window-collapse.ts`
- Modify: `tests/note-window-collapse.test.ts`

当前控制器只有 `isCollapsed` boolean。改成：

```ts
type NoteWindowPresentation = 'expanded' | 'collapsed' | 'docked';
```

公开接口扩成：

```ts
{
  getPresentation: () => NoteWindowPresentation;
  getBoundsForPersistence: () => NativeNoteWindowBounds;
  getDockForPersistence: () => { side: 'left' | 'right'; y: number } | undefined;
  setCollapsed: (collapsed: boolean) => Promise<void>;
  setDocked: (dock: { side: 'left' | 'right'; y: number } | null) => Promise<void>;
  applyRestoredDock: (dock: { side: 'left' | 'right'; y: number }) => void;
}
```

行为（与设计一致）：

- `setCollapsed(true)` 仅从 `expanded` 进入 `collapsed`。若当前 `docked`，先当作错误拒绝或先忽略——产品不允许贴边→横条，直接 `return`。
- `setCollapsed(false)` 仅从 `collapsed` 展开（现有逻辑）。`docked` 时不要走这条「用横条左上角当展开锚点」的旧路径。
- `setDocked({ side, y })` 仅从 `collapsed` 进入 `docked`：`setMinimumSize(8, 56)` → `setBounds(buildDockedBounds(...), false)` → `setResizable(false)`。失败则回滚到进入前的横条矩形。
- `setDocked(null)` 仅从 `docked` 进入 `expanded`：先 `setResizable(true)`，再 `setBounds(目标展开矩形, false)`，再 `setMinimumSize(NOTE_MIN_WIDTH, NOTE_MIN_HEIGHT)`。目标矩形由调用方传入更干净——把签名改成：

```ts
setDocked: (
  next:
    | { kind: 'dock'; side: 'left' | 'right'; y: number }
    | { kind: 'expand'; bounds: NativeNoteWindowBounds }
    | { kind: 'snap-back' }
) => Promise<void>;
```

`snap-back` 把窗口设回进入这次拖动前保存的贴边矩形。

- `getBoundsForPersistence`：`expanded` 返回当前 bounds；`collapsed` 保持现逻辑（展开宽高 + 横条 x/y）；`docked` **返回进入贴边前记住的展开 bounds，不要把 8×56 或缝的 x/y 写进 bounds**。
- `getDockForPersistence`：仅 `docked` 时返回 `{ side, y }`，y 用当前缝上沿。
- `applyRestoredDock`：启动时窗口已按缝的尺寸创建，只同步内部 `presentation`、记住 `expandedBounds`（来自 `note.bounds`）和 `dockSide`，不再 `setBounds`。

把 harness 的 `setBounds` 失败判定从「height === 40」改成按操作名注入，否则 8px 高会误判。

测试至少覆盖：

1. collapsed → docked 成功，persistence bounds 仍是展开尺寸，dock 为 `{ side, y }`
2. docked → expand 成功，presentation 为 expanded，dock 为空
3. docked snap-back 回到原缝
4. docked 时 `setBounds` 失败会回滚，可重试
5. docked 时 `setCollapsed(true/false)` 不改变状态

关键约束：`.drag-bar` 是 `-webkit-app-region: drag`，**不要**在 renderer 里监听 drag-bar 的 `pointerup` 来贴边。

在 `createManagedNoteWindow` 里：

1. `let dockDragOrigin: Rect | undefined`
2. `noteWindow.on('will-move', () => { dockDragOrigin = noteWindow.getBounds(); })`（若当前平台没有 `will-move`，用第一次 `move` 记录 origin）
3. `noteWindow.on('moved', async () => { ... })`：
   - `collapsed`：`resolveCollapsedDockSide(current, nearestWorkArea)` 有值则 `webContents.send('sticky-notes:dock-offer', { side, y })`
   - `docked`：用 origin+current 调 `resolveDockedRelease`；`snap-back` 则 `setDocked({ kind: 'snap-back' })`；`expand` 则先算 `buildExpandBoundsFromDock`，`send('sticky-notes:undock-offer', { bounds })`
4. renderer 收到 offer 后播视觉，再 `invoke` 确认：
   - `sticky-notes:accept-dock` `{ side, y }`
   - `sticky-notes:accept-undock` `{ bounds }`
5. 主进程在 accept 里调用 controller，成功后 `persistDock`

preload：

```ts
onDockOffer: (listener: (payload: { side: 'left' | 'right'; y: number }) => void) =>
  ipcRenderer.on('sticky-notes:dock-offer', (_e, payload) => listener(payload)),
onUndockOffer: (listener: (payload: { bounds: Required<NoteBounds> }) => void) =>
  ipcRenderer.on('sticky-notes:undock-offer', (_e, payload) => listener(payload)),
acceptDock: (payload: { side: 'left' | 'right'; y: number }) =>
  ipcRenderer.invoke('sticky-notes:accept-dock', payload),
acceptUndock: (payload: { bounds: Required<NoteBounds> }) =>
  ipcRenderer.invoke('sticky-notes:accept-undock', payload)
```

`getCurrentNote` 返回的 `NoteView` 已含 `dock?`。renderer 启动时若 `note.dock` 存在，初始 `presentation` 设为 `docked`，不要先闪完整便签再吸回去。

**横条 → 贴边：** 先加 `note-shell--dock-transitioning`，壳体视觉收到 8×56，等 240ms（现有 `waitForHeightTransition` / 320ms fallback），再 `acceptDock`。主进程此时才 `setBounds` 成缝。
